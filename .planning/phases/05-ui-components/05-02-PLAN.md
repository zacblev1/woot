       ```typescript
       function HistoryDisplayComponent({ history, className = '' }: HistoryDisplayProps) {
         const scrollRef = useRef<HTMLDivElement>(null)

         // Auto-scroll to bottom when history changes
         useEffect(() => {
           if (scrollRef.current) {
             scrollRef.current.scrollTop = scrollRef.current.scrollHeight
           }
         }, [history])

         return (
           <div
             ref={scrollRef}
             className={`flex-1 overflow-y-auto ${className}`}
           >
             {history.map((line, i) => (
               <TerminalLine key={i} line={line} />
             ))}
           </div>
         )
       }
       ```

    6. Wrap with React.memo:
       ```typescript
       export const HistoryDisplay = memo(HistoryDisplayComponent)
       ```

    7. Add displayName for React DevTools:
       ```typescript
       HistoryDisplay.displayName = 'HistoryDisplay'
       ```

    Key points:
    - Uses index as key (acceptable: history is append-only)
    - Memoized so it only re-renders when history array reference changes
    - Auto-scroll triggers on history change, not on every render
    - Accepts optional className for styling flexibility

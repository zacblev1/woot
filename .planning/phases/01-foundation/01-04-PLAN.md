import { describe, it, expect, beforeEach } from 'vitest'
import { VirtualFileSystem } from '../vfs'

describe('VirtualFileSystem - Characterization Tests', () => {
  let vfs: VirtualFileSystem

  beforeEach(() => {
    vfs = new VirtualFileSystem()
  })

  describe('Initial State', () => {
    it('starts in home directory /home/zachary', () => {
      expect(vfs.getPwd()).toBe('/home/zachary')
    })

    it('root directory exists', () => {
      const root = vfs.resolve('/')
      expect(root).not.toBeNull()
      expect(root?.type).toBe('directory')
    })

    it('home directory exists', () => {
      const home = vfs.resolve('/home/zachary')
      expect(home).not.toBeNull()
      expect(home?.type).toBe('directory')
    })
  })

  describe('cd - Change Directory', () => {
    it('cd to existing directory returns null (no error)', () => {
      vfs.createDir('/home/zachary/test')
      const result = vfs.cd('test')
      expect(result).toBeNull()
    })

    it('cd to non-existent directory returns error message', () => {
      const result = vfs.cd('nonexistent')
      expect(result).toBe('cd: nonexistent: No such file or directory')
    })

    it('cd to file returns error message', () => {
      vfs.touch('myfile')
      const result = vfs.cd('myfile')
      expect(result).toBe('cd: myfile: Not a directory')
    })

    it('cd .. moves to parent directory', () => {
      vfs.createDir('/home/zachary/nested')
      vfs.cd('nested')
      expect(vfs.getPwd()).toBe('/home/zachary/nested')
      vfs.cd('..')
      expect(vfs.getPwd()).toBe('/home/zachary')
    })

    it('cd ~ returns to home', () => {
      vfs.cd('/')
      expect(vfs.getPwd()).toBe('/')
      vfs.cd('~')
      expect(vfs.getPwd()).toBe('/home/zachary')
    })

    it('cd with absolute path works', () => {
      vfs.cd('/')
      expect(vfs.getPwd()).toBe('/')
    })

    it('cd at root with .. stays at root', () => {
      vfs.cd('/')
      vfs.cd('..')
      expect(vfs.getPwd()).toBe('/')
    })
  })

  describe('ls - List Directory', () => {
    it('ls on empty directory returns empty array', () => {
      const result = vfs.ls()
      expect(result).toEqual([])
    })

    it('ls returns sorted directory contents', () => {
      vfs.touch('zebra')
      vfs.touch('apple')
      vfs.mkdir('banana')
      const result = vfs.ls()
      expect(result).toEqual(['apple', 'banana', 'zebra'])
    })

    it('ls on non-existent path returns error', () => {
      const result = vfs.ls('nonexistent')
      expect(result).toEqual(['ls: nonexistent: No such file or directory'])
    })

    it('ls on file returns filename', () => {
      vfs.touch('myfile')
      const result = vfs.ls('myfile')
      expect(result).toEqual(['myfile'])
    })

    it('ls with path argument works', () => {
      vfs.createDir('/home/zachary/subdir')
      vfs.cd('subdir')
      vfs.touch('nested-file')
      vfs.cd('..')
      const result = vfs.ls('subdir')
      expect(result).toEqual(['nested-file'])
    })
  })

  describe('mkdir - Make Directory', () => {
    it('mkdir creates directory', () => {
      const error = vfs.mkdir('newdir')
      expect(error).toBeNull()
      const result = vfs.ls()
      expect(result).toContain('newdir')
    })

    it('mkdir on existing name returns error', () => {
      vfs.mkdir('existing')
      const error = vfs.mkdir('existing')
      expect(error).toBe("mkdir: cannot create directory 'existing': File exists")
    })

    it('mkdir with nested path creates in parent', () => {
      vfs.mkdir('parent')
      vfs.mkdir('parent/child')
      const result = vfs.ls('parent')
      expect(result).toContain('child')
    })

    it('mkdir with non-existent parent returns error', () => {
      const error = vfs.mkdir('nonexistent/child')
      expect(error).toBe('mkdir: nonexistent: No such file or directory')
    })
  })

  describe('touch - Create File', () => {
    it('touch creates file', () => {
      const error = vfs.touch('newfile')
      expect(error).toBeNull()
      const result = vfs.ls()
      expect(result).toContain('newfile')
    })

    it('touch on existing file returns null (updates timestamp)', () => {
      vfs.touch('existing')
      const error = vfs.touch('existing')
      expect(error).toBeNull()
    })

    it('touch with nested path creates in parent', () => {
      vfs.mkdir('parent')
      vfs.touch('parent/child.txt')
      const result = vfs.ls('parent')
      expect(result).toContain('child.txt')
    })
  })

  describe('rm - Remove', () => {
    it('rm removes file', () => {
      vfs.touch('toremove')
      expect(vfs.ls()).toContain('toremove')
      const error = vfs.rm('toremove')
      expect(error).toBeNull()
      expect(vfs.ls()).not.toContain('toremove')
    })

    it('rm removes directory', () => {
      vfs.mkdir('toremove')
      const error = vfs.rm('toremove')
      expect(error).toBeNull()
      expect(vfs.ls()).not.toContain('toremove')
    })

    it('rm on non-existent returns error', () => {
      const error = vfs.rm('nonexistent')
      expect(error).toBe('rm: nonexistent: No such file or directory')
    })

    it('rm on current directory returns error', () => {
      vfs.mkdir('current')
      vfs.cd('current')
      const error = vfs.rm('.')
      expect(error).toBe('rm: cannot remove current directory')
    })

    it('rm on root returns error', () => {
      const error = vfs.rm('/')
      expect(error).toBe('rm: cannot remove root directory')
    })
  })

  describe('resolve - Path Resolution', () => {
    it('resolve empty string returns current directory', () => {
      const result = vfs.resolve('')
      expect(result?.name).toBe('zachary')
    })

    it('resolve ~ returns home', () => {
      const result = vfs.resolve('~')
      expect(result?.name).toBe('zachary')
    })

    it('resolve ~/ path works', () => {
      vfs.mkdir('subdir')
      const result = vfs.resolve('~/subdir')
      expect(result?.name).toBe('subdir')
    })

    it('resolve . returns current', () => {
      const result = vfs.resolve('.')
      expect(result).toBe(vfs.resolve(''))
    })

    it('resolve .. returns parent', () => {
      const result = vfs.resolve('..')
      expect(result?.name).toBe('home')
    })

    it('resolve non-existent returns null', () => {
      const result = vfs.resolve('nonexistent')
      expect(result).toBeNull()
    })
  })

  describe('Serialization', () => {
    it('toJSON produces valid JSON string', () => {
      vfs.mkdir('testdir')
      vfs.touch('testfile')
      const json = vfs.toJSON()
      expect(() => JSON.parse(json)).not.toThrow()
    })

    it('fromJSON restores filesystem state', () => {
      vfs.mkdir('persistdir')
      vfs.touch('persistfile')
      const json = vfs.toJSON()

      const newVfs = new VirtualFileSystem()
      newVfs.fromJSON(json)

      expect(newVfs.ls('/home/zachary')).toContain('persistdir')
      expect(newVfs.ls('/home/zachary')).toContain('persistfile')
    })

    it('fromJSON restores to home directory', () => {
      vfs.cd('/')
      const json = vfs.toJSON()

      const newVfs = new VirtualFileSystem()
      newVfs.fromJSON(json)

      expect(newVfs.getPwd()).toBe('/home/zachary')
    })
  })
})

import { describe, it, expect, vi, beforeEach, afterEach } from 'vitest'
import { render, screen, fireEvent, waitFor } from '@testing-library/react'
import userEvent from '@testing-library/user-event'
import { Terminal } from '../terminal'

// Mock localStorage
const localStorageMock = (() => {
  let store: Record<string, string> = {}
  return {
    getItem: vi.fn((key: string) => store[key] || null),
    setItem: vi.fn((key: string, value: string) => { store[key] = value }),
    removeItem: vi.fn((key: string) => { delete store[key] }),
    clear: vi.fn(() => { store = {} }),
  }
})()

Object.defineProperty(window, 'localStorage', { value: localStorageMock })

// Mock matchMedia for responsive testing
Object.defineProperty(window, 'matchMedia', {
  value: vi.fn().mockImplementation(query => ({
    matches: false,
    media: query,
    onchange: null,
    addListener: vi.fn(),
    removeListener: vi.fn(),
    addEventListener: vi.fn(),
    removeEventListener: vi.fn(),
    dispatchEvent: vi.fn(),
  })),
})

describe('Terminal - Characterization Tests', () => {
  beforeEach(() => {
    localStorageMock.clear()
    vi.clearAllMocks()
  })

  describe('Initial Render', () => {
    it('renders terminal container', () => {
      render(<Terminal />)
      // Terminal should have input field
      const input = screen.getByRole('textbox')
      expect(input).toBeInTheDocument()
    })

    it('shows welcome message on load', async () => {
      render(<Terminal />)
      // Wait for initial render
      await waitFor(() => {
        expect(screen.getByText(/welcome/i)).toBeInTheDocument()
      })
    })

    it('shows prompt with home directory path', async () => {
      render(<Terminal />)
      await waitFor(() => {
        expect(screen.getByText(/~\/\s*\$/)).toBeInTheDocument()
      })
    })
  })

  describe('Command Input', () => {
    it('accepts text input', async () => {
      const user = userEvent.setup()
      render(<Terminal />)

      const input = screen.getByRole('textbox')
      await user.type(input, 'ls')

      expect(input).toHaveValue('ls')
    })

    it('clears input after Enter', async () => {
      const user = userEvent.setup()
      render(<Terminal />)

      const input = screen.getByRole('textbox')
      await user.type(input, 'pwd{Enter}')

      expect(input).toHaveValue('')
    })

    it('shows command in history after execution', async () => {
      const user = userEvent.setup()
      render(<Terminal />)

      const input = screen.getByRole('textbox')
      await user.type(input, 'pwd{Enter}')

      await waitFor(() => {
        // The command should appear in history
        expect(screen.getByText(/pwd/)).toBeInTheDocument()
      })
    })
  })

  describe('Basic Commands', () => {
    it('pwd shows current directory', async () => {
      const user = userEvent.setup()
      render(<Terminal />)

      const input = screen.getByRole('textbox')
      await user.type(input, 'pwd{Enter}')

      await waitFor(() => {
        expect(screen.getByText('/home/zachary')).toBeInTheDocument()
      })
    })

    it('help shows available commands', async () => {
      const user = userEvent.setup()
      render(<Terminal />)

      const input = screen.getByRole('textbox')
      await user.type(input, 'help{Enter}')

      await waitFor(() => {
        expect(screen.getByText(/available commands/i)).toBeInTheDocument()
      })
    })

    it('clear removes history', async () => {
      const user = userEvent.setup()
      render(<Terminal />)

      const input = screen.getByRole('textbox')
      await user.type(input, 'pwd{Enter}')

      // Wait for pwd output
      await waitFor(() => {
        expect(screen.getByText('/home/zachary')).toBeInTheDocument()
      })

      await user.type(input, 'clear{Enter}')

      // pwd output should be gone (only welcome remains or is also cleared)
      await waitFor(() => {
        const pwdOutputs = screen.queryAllByText('/home/zachary')
        // After clear, there might be no pwd output visible
        expect(pwdOutputs.length).toBeLessThanOrEqual(1)
      })
    })

    it('unknown command shows error', async () => {
      const user = userEvent.setup()
      render(<Terminal />)

      const input = screen.getByRole('textbox')

      await waitFor(() => {
        expect(screen.getByText(/command not found/i)).toBeInTheDocument()

        // Trivia should show a question
        expect(screen.getByText(/\?|question|trivia/i)).toBeInTheDocument()

      const user = userEvent.setup()

      await user.type(input, 'game blackjack{Enter}')

      await waitFor(() => {
        // Blackjack should show cards or betting
        expect(screen.getByText(/card|hit|stand|blackjack/i)).toBeInTheDocument()
      })
    })

    it('game rps starts without error', async () => {
      const user = userEvent.setup()
      render(<Terminal />)

      const input = screen.getByRole('textbox')
      await user.type(input, 'game rps{Enter}')

      await waitFor(() => {
        // RPS should show choices
        expect(screen.getByText(/rock|paper|scissors/i)).toBeInTheDocument()
      })
    })

    // Note: Tron test is more complex due to canvas rendering
    // We verify it loads without crashing via dynamic import
    it('game tron command is recognized', async () => {
      const user = userEvent.setup()
      render(<Terminal />)

      const input = screen.getByRole('textbox')
      await user.type(input, 'game tron{Enter}')

      // Tron should either show loading or the game
      await waitFor(() => {
        // Either loading message or game canvas should appear
        const hasLoadingOrGame =
          screen.queryByText(/loading tron/i) ||
          screen.queryByText(/tron/i) ||
          document.querySelector('canvas')
        expect(hasLoadingOrGame).toBeTruthy()
      }, { timeout: 3000 })
    })
  })

  describe('Theme Persistence', () => {
    it('saves theme to localStorage', async () => {
      const user = userEvent.setup()
      render(<Terminal />)

      const input = screen.getByRole('textbox')
      await user.type(input, 'theme dracula{Enter}')

      await waitFor(() => {
        expect(localStorageMock.setItem).toHaveBeenCalledWith(
          'terminal-theme',
          expect.stringContaining('dracula')
        )
      })
    })

    it('loads theme from localStorage on mount', async () => {
      localStorageMock.getItem.mockImplementation((key: string) => {
        if (key === 'terminal-theme') return 'nord'
        return null
      })

      render(<Terminal />)

      // Theme should be loaded (we can't easily test CSS vars, but no crash means success)
      expect(localStorageMock.getItem).toHaveBeenCalledWith('terminal-theme')
    })
  })

  describe('Font Persistence', () => {
    it('saves font to localStorage', async () => {
      const user = userEvent.setup()
      render(<Terminal />)

      const input = screen.getByRole('textbox')
      await user.type(input, 'font fira{Enter}')

      await waitFor(() => {
        expect(localStorageMock.setItem).toHaveBeenCalledWith(
          'terminal-font',
          expect.stringContaining('fira')
        )
      })
    })
  })

  describe('VFS Persistence', () => {
    it('saves VFS changes to localStorage', async () => {
      const user = userEvent.setup()
      render(<Terminal />)

      const input = screen.getByRole('textbox')
      await user.type(input, 'mkdir testdir{Enter}')

      await waitFor(() => {
        expect(localStorageMock.setItem).toHaveBeenCalledWith(
          'vfs',
          expect.any(String)
        )
      })
    })
  })

  describe('Keyboard Navigation', () => {
    it('Ctrl+C during game shows interrupt message', async () => {
      const user = userEvent.setup()
      render(<Terminal />)

      const input = screen.getByRole('textbox')
      await user.type(input, 'game number{Enter}')

      // Wait for game to start
      await waitFor(() => {
        expect(screen.getByText(/guess/i)).toBeInTheDocument()
      })

      // Send Ctrl+C
      fireEvent.keyDown(input, { key: 'c', ctrlKey: true })

      await waitFor(() => {
        // Should return to normal prompt or show interrupt
        expect(screen.getByText(/interrupted|quit|\$/)).toBeInTheDocument()
      })
    })

    it('Up arrow navigates command history', async () => {
      const user = userEvent.setup()
      render(<Terminal />)

      const input = screen.getByRole('textbox')

      // Execute a command
      await user.type(input, 'pwd{Enter}')

      // Press up arrow
      await user.type(input, '{ArrowUp}')

      // Input should show previous command
      expect(input).toHaveValue('pwd')
    })
  })
})

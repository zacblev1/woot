```typescript
'use client'

import { createContext, useContext, useState, useEffect, useCallback, type ReactNode } from 'react'
import { VirtualFileSystem } from '@/lib/vfs'
import type { VFSNode } from '@/lib/types'

// Storage key for VFS persistence
const VFS_STORAGE_KEY = 'terminal-vfs'

// Context value type
export interface VFSContextValue {
  // Current working directory path
  pwd: string

  // Navigation
  cd: (path: string) => string | null

  // Directory listing
  ls: (path?: string) => string[]

  // Path resolution
  resolve: (path: string) => VFSNode | null

  // File operations
  mkdir: (path: string) => string | null
  touch: (path: string) => string | null
  rm: (path: string) => string | null

  // Persistence
  save: () => void

  // Direct VFS access (for collection initialization)
  vfs: VirtualFileSystem
}

const VFSContext = createContext<VFSContextValue | null>(null)

interface VFSProviderProps {
  children: ReactNode
  initialData?: {
    books?: unknown[]
    vinyl?: unknown[]
    hardware?: unknown[]
  }
}

export function VFSProvider({ children, initialData }: VFSProviderProps) {
  const [vfs] = useState(() => new VirtualFileSystem())
  const [pwd, setPwd] = useState(() => vfs.getPwd())
  const [, forceUpdate] = useState(0)

  // Initialize VFS from localStorage or with initial data
  useEffect(() => {
    if (typeof window === 'undefined') return

    const saved = localStorage.getItem(VFS_STORAGE_KEY)
    if (saved) {
      try {
        vfs.fromJSON(saved)
        setPwd(vfs.getPwd())
      } catch (e) {
        console.error('Failed to restore VFS from localStorage', e)
      }
    }

    // Initialize collection directories if provided
    // This will be handled by the terminal component for now
    // to maintain backward compatibility
  }, [vfs])

  // Save VFS to localStorage
  const save = useCallback(() => {
    if (typeof window === 'undefined') return
    try {
      localStorage.setItem(VFS_STORAGE_KEY, vfs.toJSON())
    } catch (e) {
      console.error('Failed to save VFS to localStorage', e)
    }
  }, [vfs])

  // Wrapped cd that updates pwd state
  const cd = useCallback((path: string): string | null => {
    const error = vfs.cd(path)
    if (!error) {
      setPwd(vfs.getPwd())
      save()
    }
    return error
  }, [vfs, save])

  // Wrapped mkdir that triggers save
  const mkdir = useCallback((path: string): string | null => {
    const error = vfs.mkdir(path)
    if (!error) {
      save()
      forceUpdate(n => n + 1)
    }
    return error
  }, [vfs, save])

  // Wrapped touch that triggers save
  const touch = useCallback((path: string): string | null => {
    const error = vfs.touch(path)
    if (!error) {
      save()
      forceUpdate(n => n + 1)
    }
    return error
  }, [vfs, save])

  // Wrapped rm that triggers save
  const rm = useCallback((path: string): string | null => {
    const error = vfs.rm(path)
    if (!error) {
      save()
      forceUpdate(n => n + 1)
    }
    return error
  }, [vfs, save])

  // Direct passthrough methods
  const ls = useCallback((path?: string) => vfs.ls(path), [vfs])
  const resolve = useCallback((path: string) => vfs.resolve(path) as VFSNode | null, [vfs])

  const value: VFSContextValue = {
    pwd,
    cd,
    ls,
    resolve,
    mkdir,
    touch,
    rm,
    save,
    vfs,
  }

  return (
    <VFSContext.Provider value={value}>
      {children}
    </VFSContext.Provider>
  )
}

export function useVFS(): VFSContextValue {
  const context = useContext(VFSContext)
  if (!context) {
    throw new Error('useVFS must be used within a VFSProvider')
  }
  return context
}

// Optional hook that returns null instead of throwing (for conditional usage)
export function useVFSOptional(): VFSContextValue | null {
  return useContext(VFSContext)
}
```

Key design decisions:
- 'use client' directive for Next.js App Router compatibility
- VFS instance created once via useState initializer
- pwd tracked as React state for reactivity
- Operations wrapped to trigger persistence
- Helpful error message when used outside provider
- Optional hook for conditional usage

```typescript
import { describe, it, expect, vi, beforeEach } from 'vitest'
import { render, screen, act } from '@testing-library/react'
import { VFSProvider, useVFS } from '../VFSContext'

// Mock localStorage
const localStorageMock = {
  getItem: vi.fn(),
  setItem: vi.fn(),
  clear: vi.fn(),
}
Object.defineProperty(window, 'localStorage', { value: localStorageMock })

// Test component that uses the hook
function TestConsumer() {
  const { pwd, cd, ls, mkdir } = useVFS()
  return (
    <div>
      <div data-testid="pwd">{pwd}</div>
      <button onClick={() => cd('books')}>cd books</button>
      <button onClick={() => mkdir('test-dir')}>mkdir</button>
      <div data-testid="ls">{ls().join(', ')}</div>
    </div>
  )
}

describe('VFSContext', () => {
  beforeEach(() => {
    vi.clearAllMocks()
    localStorageMock.getItem.mockReturnValue(null)
  })

  it('provides VFS context to children', () => {
    render(
      <VFSProvider>
        <TestConsumer />
      </VFSProvider>
    )

    expect(screen.getByTestId('pwd')).toHaveTextContent('/home/zachary')
  })

  it('throws when useVFS is called outside provider', () => {
    // Suppress console.error for this test
    const consoleSpy = vi.spyOn(console, 'error').mockImplementation(() => {})

    expect(() => {
      render(<TestConsumer />)
    }).toThrow('useVFS must be used within a VFSProvider')

    consoleSpy.mockRestore()
  })

  it('cd updates pwd state', async () => {
    render(
      <VFSProvider>
        <TestConsumer />
      </VFSProvider>
    )

    // Initial state
    expect(screen.getByTestId('pwd')).toHaveTextContent('/home/zachary')

    // Navigate (books dir created during VFS init from terminal)
    // For this test, we need to cd to an existing dir like '..'
    await act(async () => {
      screen.getByText('cd books').click()
    })

    // Since books doesn't exist yet in base VFS, it should show error
    // But pwd shouldn't change on failed cd
    expect(screen.getByTestId('pwd')).toHaveTextContent('/home/zachary')
  })

  it('mkdir creates directory and saves to localStorage', async () => {
    render(
      <VFSProvider>
        <TestConsumer />
      </VFSProvider>
    )

    await act(async () => {
      screen.getByText('mkdir').click()
    })

    expect(localStorageMock.setItem).toHaveBeenCalled()
  })

  it('ls returns directory contents', () => {
    render(
      <VFSProvider>
        <TestConsumer />
      </VFSProvider>
    )

    // Home directory should exist (might be empty)
    const lsResult = screen.getByTestId('ls')
    expect(lsResult).toBeInTheDocument()
  })
})
```

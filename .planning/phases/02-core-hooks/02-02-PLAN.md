```typescript
// Maps game type to its data type
export type GameDataForType<T extends GameType> =
  T extends 'number' ? NumberGameData :
  T extends 'wordle' ? WordleGameData :
  T extends 'trivia' ? TriviaGameData :
  T extends 'blackjack' ? BlackjackGameData :
  T extends 'rps' ? RPSGameData :
  T extends 'tron' ? TronGameData :
  never
```

**Interface (export as type):**
```typescript
export interface UseGameStateReturn {
  gameState: GameState
  isPlaying: boolean
  currentGame: GameType | null
  startGame: <T extends GameType>(type: T, initialData: GameDataForType<T>) => void
  endGame: () => void
  updateData: <T extends GameType>(updater: (current: GameDataForType<T>) => GameDataForType<T>) => void
}
```

**Implementation details:**
- Initialize with `createInactiveGameState()` from types/games.ts
- `isPlaying`: derived from `gameState.active`
- `currentGame`: derived from `gameState.active ? gameState.type : null`
- `startGame<T>`: Set state to `{ active: true, type, data: initialData }`
- `endGame`: Set state to `createInactiveGameState()`
- `updateData`: If not active, return early. Otherwise use functional update: `{ ...prev, data: updater(prev.data) }`

**Important:**
- Wrap startGame, endGame, updateData in `useCallback` with empty deps (use functional updates)
- The updateData function should guard against inactive state (no-op if not playing)
- Use type assertion when needed: `updater(prev.data as GameDataForType<T>)`
